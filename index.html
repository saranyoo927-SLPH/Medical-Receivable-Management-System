/** * Backend สำหรับ MRMS IPD Dashboard * ผูกสคริปต์นี้กับ Google Sheets ไฟล์ทะเบียนลูกหนี้ แล้ว Deploy > New deployment > Web app * Execute as: Me * Who has access: Anyone * จากนั้นนำ URL /exec ไปใส่ใน DATA_URL ของไฟล์ HTML */ const TARGET_SHEET_NAME = 'ทะเบียนคุมลูกหนี้'; // ใช้ trim() จึงไม่ต้องกังวลช่องว่างท้ายชื่อชีต function doGet() { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss) throw new Error('ไม่พบ Spreadsheet ที่ผูกกับ Apps Script'); const sheet = ss.getSheets().find(s => s.getName().trim() === TARGET_SHEET_NAME); if (!sheet) { throw new Error('ไม่พบชีต "ทะเบียนคุมลูกหนี้" กรุณาตรวจสอบชื่อชีต'); } const lastRow = sheet.getLastRow(); if (lastRow < 3) return jsonOutput({ ok: true, rows: [], asOf: formatNow() }); // อ่าน A:AE เพราะข้อมูลที่ Dashboard ใช้กระจายอยู่ถึงคอลัมน์ AE // แถว 1 เป็นสรุปยอด และแถว 2 เป็นหัวตารางจริง จึงเริ่มอ่านที่แถว 2 const values = sheet.getRange(2, 1, lastRow - 1, 31).getValues(); const headers = values[0].map(v => String(v ?? '').trim()); // ตรวจโครงสร้างเพื่อให้พบความผิดพลาดเร็ว หากมีการย้ายคอลัมน์ในอนาคต const required = [ 'ภาระหนี้ รอชำระ', 'ประมาณการชดเชย', 'รับชำระลูกหนี้', 'ลูกหนี้คงเหลือ', 'เดือนตั้งหนี้', 'สถานะการชำระเงิน', 'จำนวนวัน', 'สถานะการเคลม (ส่งผ่าน eclaim.)' ]; const missing = required.filter(h => !headers.includes(h)); if (missing.length) { throw new Error('หัวตารางไม่ตรง: ไม่พบ ' + missing.join(', ')); } const rows = []; for (let i = 1; i < values.length; i++) { const r = values[i]; // ตำแหน่งคอลัมน์ตามชีตปัจจุบัน: // F=5, I=8, K=10, R=17, T=19, V=21, X=23, Y=24, AB=27, AE=30 (index เริ่ม 0) const record = { 'สิทธรักษา': safeText(r[24]) || safeText(r[5]), 'ภาระหนี้ รอชำระ': safeNumber(r[8]), 'ประมาณการชดเชย': safeNumber(r[10]), 'รับชำระลูกหนี้': safeNumber(r[17]), 'ลูกหนี้คงเหลือ': safeNumber(r[19]), 'เดือนตั้งหนี้': safeText(r[21]), 'สถานะการชำระเงิน': safeText(r[23]), 'จำนวนวัน': safeText(r[27]), 'สถานะการเคลม (ส่งผ่าน eclaim.)': safeText(r[30]) }; const hasText = [ record['สิทธรักษา'], record['เดือนตั้งหนี้'], record['สถานะการชำระเงิน'], record['จำนวนวัน'], record['สถานะการเคลม (ส่งผ่าน eclaim.)'] ].some(Boolean); const hasMoney = [ record['ภาระหนี้ รอชำระ'], record['ประมาณการชดเชย'], record['รับชำระลูกหนี้'], record['ลูกหนี้คงเหลือ'] ].some(v => v !== 0); if (hasText || hasMoney) rows.push(record); } return jsonOutput({ ok: true, sheet: sheet.getName(), rowCount: rows.length, asOf: formatNow(), rows }); } catch (err) { return jsonOutput({ ok: false, error: err && err.message ? err.message : String(err), asOf: formatNow(), rows: [] }); } } function safeText(v) { if (v === null || v === undefined) return ''; const s = String(v).trim(); return s === '#REF!' ? 'ต้องตรวจสอบสูตรอายุหนี้' : s; } function safeNumber(v) { if (typeof v === 'number' && isFinite(v)) return v; const n = Number(String(v ?? '').replace(/,/g, '').trim()); return isFinite(n) ? n : 0; } function jsonOutput(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function formatNow() { return Utilities.formatDate(new Date(), 'Asia/Bangkok', 'dd/MM/yyyy HH:mm'); }
